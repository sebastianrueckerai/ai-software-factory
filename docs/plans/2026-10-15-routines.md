> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Recurring work becomes data per project ("routines"), its clock becomes Temporal Schedules and store counters reconciled from the configuration tree, each run files a card through the ordinary intake, every routine stands as a ghost card at the foot of its lane on the board, and the chat creates, changes and deletes routines through tools.

**Architecture:** A `routines:` block in a project's file (and in `policy.yaml` for the template pass) is parsed and validated by `internal/policy` like every other block. A new `factoryd.Routines` service reconciles the tree into Temporal Schedules (one per cron trigger), counts merges and events in the store for the other triggers, and starts one `RoutineRunWorkflow` per due run. The run's one activity files the card by kind through the existing `FileTask`, `FileAgentTask` and the template pass, writes the run's row, and pokes the inbox. The console reads the routines' views through a port, draws the ghosts, the strip cell, the chip, the provenance line and the routine's drawer, and answers the two acts. The chat's tools edit the project's file through the same commit path as the autonomy level.

**Tech Stack:** Go 1.25, Temporal Go SDK v1.44 (`client.ScheduleClient`), PostgreSQL with hand-written SQL, `github.com/robfig/cron/v3` for cron parsing, Go templates with htmx, Playwright for the console lab.

```go
package policy

import (
	"fmt"
	"regexp"
	"strings"
	"time"

	"github.com/robfig/cron/v3"
)

// A routine is a named piece of recurring work with a schedule (roadmap 267, C16). Its run files a card.

// Trigger is one entry of a routine's schedule: a cron pattern, a merge count or an event. Exactly one field is set.
type Trigger struct {
	// Cron is five fields, minute hour day-of-month month day-of-week, in the factory's time zone.
	Cron string `json:"cron,omitempty"`
	// Merges runs the routine every n merged cards of the project.
	Merges int `json:"merges,omitempty"`
	// On runs the routine on an event: configuration-version or push-main.
	On string `json:"on,omitempty"`
}

// Routine is recurring work the factory does on its own: what it files, when, and the rules of a run.
type Routine struct {
	// Name is the routine's slug, unique in its file.
	Name string `json:"name"`
	// Title is the filed card's title.
	Title string `json:"title"`
	// Schedule lists the triggers; any one of them starts a run.
	Schedule []Trigger `json:"schedule"`
	// Files is the kind of work a run files: human, agent, pull-request or report.
	Files string `json:"files"`
	// Brief is the path of the card's body in the tree, routines/<file>.md.
	Brief string `json:"brief,omitempty"`
	// Skill is the skill an agent card runs.
	Skill string `json:"skill,omitempty"`
	// Pass names the pass a pull-request routine runs: template.
	Pass string `json:"pass,omitempty"`
	// WhileOpen says what a run does while the last card is open: hold (the default), fold or file.
	WhileOpen string `json:"while_open,omitempty"`
	// Missed says what a run does when its time passed while the factory was down: file-once (the default) or skip.
	Missed string `json:"missed,omitempty"`
	// Tell says whether a filing pokes the inbox: inbox (the default) or quiet.
	Tell string `json:"tell,omitempty"`
}

const (
	FilesHuman       = "human"
	FilesAgent       = "agent"
	FilesPullRequest = "pull-request"
	FilesReport      = "report"

	OnConfigurationVersion = "configuration-version"
	OnPushMain             = "push-main"

	WhileOpenHold = "hold"
	WhileOpenFold = "fold"
	WhileOpenFile = "file"
	MissedFileOnce = "file-once"
	MissedSkip     = "skip"
	TellInbox      = "inbox"
	TellQuiet      = "quiet"
	PassTemplate   = "template"
)

var cronParser = cron.NewParser(cron.Minute | cron.Hour | cron.Dom | cron.Month | cron.Dow)
var slug = regexp.MustCompile(`^[a-z0-9]+(-[a-z0-9]+)*$`)

func (r Routine) Validate(where string) error {
	at := fmt.Sprintf("%s: routine %q", where, r.Name)
	if !slug.MatchString(r.Name) { return fmt.Errorf("%s: the name is not a slug", at) }
	if r.Title == "" { return fmt.Errorf("%s: no title", at) }
	switch r.Files {
	case FilesHuman, FilesReport:
	case FilesAgent:
		if r.Skill == "" { return fmt.Errorf("%s: an agent routine names its skill", at) }
	case FilesPullRequest:
		if r.Pass != PassTemplate { return fmt.Errorf("%s: a pull-request routine names its pass, template", at) }
	default:
		return fmt.Errorf("%s: files is %q, not human, agent, pull-request or report", at, r.Files)
	}
	if len(r.Schedule) == 0 { return fmt.Errorf("%s: no schedule", at) }
	for _, tr := range r.Schedule {
		set := 0
		if tr.Cron != "" { set++ }
		if tr.Merges != 0 { set++ }
		if tr.On != "" { set++ }
		if set != 1 { return fmt.Errorf("%s: a trigger is one of cron, merges or on", at) }
		if tr.Cron != "" {
			if _, err := cronParser.Parse(tr.Cron); err != nil { return fmt.Errorf("%s: cron %q: %v", at, tr.Cron, err) }
		}
		if tr.Merges < 0 { return fmt.Errorf("%s: merges is negative", at) }
		if tr.On != "" && tr.On != OnConfigurationVersion && tr.On != OnPushMain { return fmt.Errorf("%s: on is %q, not configuration-version or push-main", at, tr.On) }
	}
	if r.WhileOpen != "" && r.WhileOpen != WhileOpenHold && r.WhileOpen != WhileOpenFold && r.WhileOpen != WhileOpenFile { return fmt.Errorf("%s: while_open is %q", at, r.WhileOpen) }
	if r.Missed != "" && r.Missed != MissedFileOnce && r.Missed != MissedSkip { return fmt.Errorf("%s: missed is %q", at, r.Missed) }
	if r.Tell != "" && r.Tell != TellInbox && r.Tell != TellQuiet { return fmt.Errorf("%s: tell is %q", at, r.Tell) }
	if r.Brief != "" && !strings.HasPrefix(r.Brief, "routines/") { return fmt.Errorf("%s: brief must be under routines/", at) }
	return nil
}

func (r Routine) Words() string {
	var parts []string
	for _, t := range r.Schedule { parts = append(parts, t.Words()) }
	return strings.Join(parts, " · ")
}

var weekdays = map[string]string{"0": "Sunday", "1": "Monday", "2": "Tuesday", "3": "Wednesday", "4": "Thursday", "5": "Friday", "6": "Saturday", "7": "Sunday"}

// Words states a trigger in words where the pattern is a common one, else the expression itself.
func (t Trigger) Words() string {
	switch {
	case t.Merges > 0:
		return fmt.Sprintf("every %d merges", t.Merges)
	case t.On == OnConfigurationVersion:
		return "on each configuration version"
	case t.On == OnPushMain:
		return "on each push to main"
	}
	f := strings.Fields(t.Cron)
	if len(f) != 5 { return t.Cron }
	min, hour, dom, mon, dow := f[0], f[1], f[2], f[3], f[4]
	num := func(s string) (int, bool) { var n int; _, err := fmt.Sscanf(s, "%d", &n); return n, err == nil && !strings.ContainsAny(s, "*/,-") }
	m, mok := num(min); h, hok := num(hour)
	switch {
	case mok && hour == "*" && dom == "*" && mon == "*" && dow == "*":
		return "hourly"
	case mok && hok && dom == "*" && mon == "*" && dow == "*":
		return fmt.Sprintf("daily %02d:%02d", h, m)
	case mok && hok && dom == "*" && mon == "*" && weekdays[dow] != "":
		return fmt.Sprintf("every %s %02d:%02d", weekdays[dow], h, m)
	case mok && hok && mon == "*" && dow == "*":
		if d, ok := num(dom); ok { return fmt.Sprintf("monthly on the %s %02d:%02d", ordinal(d), h, m) }
	}
	return t.Cron
}

func ordinal(d int) string {
	suffix := "th"
	switch { case d%10 == 1 && d != 11: suffix = "st"; case d%10 == 2 && d != 12: suffix = "nd"; case d%10 == 3 && d != 13: suffix = "rd" }
	return fmt.Sprintf("%d%s", d, suffix)
}

// NextCron is the earliest next time of the routine's cron triggers after now, in loc.
func (r Routine) NextCron(now time.Time, loc *time.Location) (time.Time, bool) {
	var best time.Time
	for _, t := range r.Schedule {
		if t.Cron == "" { continue }
		s, err := cronParser.Parse(t.Cron)
		if err != nil { continue }
		n := s.Next(now.In(loc))
		if best.IsZero() || n.Before(best) { best = n }
	}
	return best, !best.IsZero()
}

func (r Routine) MergesEvery() int { for _, t := range r.Schedule { if t.Merges > 0 { return t.Merges } }; return 0 }
func (r Routine) Has(on string) bool { for _, t := range r.Schedule { if t.On == on { return true } }; return false }
func (r Routine) whileOpen() string { if r.WhileOpen == "" { return WhileOpenHold }; return r.WhileOpen }
```

```go
// RoutineRun is a routine's one activity (roadmap 267): it begins the run (a second begin on the same due is a no-op),
// applies Skip once, the hold rule and the missed rule, files the card the routine's kind asks for, ends the run's
// row and pokes the inbox when the routine tells.
func (r *Runner) RoutineRun(ctx context.Context, in RoutineRunInput) (RoutineRunResult, error) {
	v := r.Current()
	rt, ok := v.Tree.Routine(in.Repository, in.Routine)
	if !ok {
		return RoutineRunResult{Outcome: "gone", Note: "the routine is no longer in the configuration"}, nil
	}
	run, began, err := r.Routines.BeginRoutineRun(ctx, RoutineRun{Repository: in.Repository, Routine: in.Routine, Due: in.Due, Trigger: in.Trigger})
	if err != nil { return RoutineRunResult{}, err }
	if !began { // the run with this due exists: BeginRoutineRun returned it
		return RoutineRunResult{Outcome: run.Outcome, Note: run.Note, TaskID: run.TaskID, Issue: run.Issue}, nil
	}
	end := func(outcome, note, taskID string, issue int64) (RoutineRunResult, error) {
		if err := r.Routines.EndRoutineRun(ctx, in.Repository, in.Routine, run.Run, outcome, note, taskID, issue, 0); err != nil { return RoutineRunResult{}, err }
		return RoutineRunResult{Outcome: outcome, Note: note, TaskID: taskID, Issue: issue}, nil
	}
	if run.Outcome == "skipped" { return end("skipped", run.Note, "", 0) }
	late := r.Now().Sub(in.Due)
	if in.Trigger == "cron" && late > 10*time.Minute && rt.MissedRule() == policy.MissedSkip {
		return end("skipped", "missed: the factory was down at "+in.Due.In(loc(v)).Format("Mon 15:04"), "", 0)
	}
	if open, ok, err := r.Routines.OpenRoutineTask(ctx, in.Repository, in.Routine); err != nil { return RoutineRunResult{}, err } else if ok {
		switch rt.WhileOpenRule() {
		case policy.WhileOpenHold:
			res, err := end("held", fmt.Sprintf("held: #%d is still open", open.IssueNumber), open.ID, open.IssueNumber)
			r.pokeRoutine(ctx, in, rt, run.Run, "The "+rt.Title+" is held: #"+strconv.FormatInt(open.IssueNumber, 10)+" from the last run is still open.")
			return res, err
		case policy.WhileOpenFold:
			return end("folded", fmt.Sprintf("folded into #%d", open.IssueNumber), open.ID, open.IssueNumber)
		}
	}
	switch rt.Files {
	case policy.FilesHuman: /* FileTask with Kind "human", RiskClass "standard", Source SourceRoutine, Author bot, body = brief + "\n\nfiled by the factory · routine " + rt.Name + " · run " + n */
	case policy.FilesAgent:  /* FileAgentTask as rolltask.go:617 files a roll: Source SourceRoutine, Objective rt.Title + brief, the skill in the brief's first line "Run the skill <skill>." until roadmap 226's pack maps skills, BudgetEUR r.Budget().ClampBudget(0) */
	case policy.FilesPullRequest: /* r.TemplatePass(ctx) -> (cards int, err); outcome "filed" with note "n cards" or "no change" */
	case policy.FilesReport: /* a poke only: outcome "reported" */
	}
	// end("filed", "", t.ID, issue) then pokeRoutine when rt.TellRule() == inbox or the card is human
}
```
